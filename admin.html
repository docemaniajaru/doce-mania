(async function () { 'use strict'; const $ = id => document.getElementById(id); const BUCKET = 'produtos'; const MAX_IMAGE_SIZE = 5 * 1024 * 1024; let db; let categories = []; let productsCache = []; let feesCache = []; let currentImageUrl = ''; let previewUrl = ''; function showStatus(message) { $('status').textContent = message; } function setLoggedIn(value) { $('loginSection').classList.toggle('hidden', value); $('panel').classList.toggle('hidden', !value); } function money(value) { return Number(value || 0).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); } function addText(parent, tag, text) { const el = document.createElement(tag); el.textContent = text; parent.appendChild(el); return el; } function resetPreview() { if (previewUrl) { URL.revokeObjectURL(previewUrl); previewUrl = ''; } const img = $('imagePreview'); if (!img) return; if (currentImageUrl) { img.src = currentImageUrl; img.classList.remove('hidden'); } else { img.removeAttribute('src'); img.classList.add('hidden'); } } function clearProductForm() { $('productForm').reset(); $('productId').value = ''; $('imageFile').value = ''; currentImageUrl = ''; $('available').checked = true; resetPreview(); showStatus('Formulário de produto limpo.'); } function clearFeeForm() { $('feeForm').reset(); $('feeId').value = ''; $('active').checked = true; } // Inicialização do Supabase const config = window.DOCE_MANIA_CONFIG || {}; const supabaseUrl = window.SUPABASE_URL || config.supabaseUrl || ''; const supabaseKey = window.SUPABASE_PUBLISHABLE_KEY || config.supabaseKey || ''; if (!window.supabase || !window.supabase.createClient) { showStatus('Biblioteca Supabase não carregou. Verifique a internet.'); return; } if (!supabaseUrl || !supabaseKey) { showStatus('Configuração ausente. Confira supabase-config.js.'); return; } try { db = window.supabase.createClient(supabaseUrl, supabaseKey); } catch (error) { showStatus('Erro ao iniciar Supabase: ' + error.message); return; } // Categorias async function loadCategories() { const { data, error } = await db .from('categories') .select('id,name,sort_order,active') .eq('active', true) .order('sort_order', { ascending: true }) .order('name', { ascending: true }); if (error) { showStatus('Erro ao carregar categorias: ' + error.message); return false; } categories = data || []; const select = $('category'); const filter = $('filterCategory'); select.replaceChildren(); filter.replaceChildren(); const placeholder = document.createElement('option'); placeholder.value = ''; placeholder.textContent = 'Selecione uma categoria'; select.appendChild(placeholder); const all = document.createElement('option'); all.value = ''; all.textContent = 'Todas as categorias'; filter.appendChild(all); categories.forEach(cat => { const option = document.createElement('option'); option.value = cat.id; option.textContent = cat.name; select.appendChild(option); const filterOption = document.createElement('option'); filterOption.value = cat.id; filterOption.textContent = cat.name; filter.appendChild(filterOption); }); if (!categories.length) { showStatus('Nenhuma categoria ativa encontrada.'); return false; } return true; } function categoryName(id) { const cat = categories.find(c => String(c.id) === String(id)); return cat ? cat.name : 'Sem categoria'; } // Verificação do administrador async function verifyAdmin() { const { data, error } = await db.auth.getUser(); if (error || !data.user) { setLoggedIn(false); showStatus('Entre com seu e-mail e sua senha do Supabase.'); return false; } const result = await db .from('admin_users') .select('user_id') .eq('user_id', data.user.id) .maybeSingle(); if (result.error) { setLoggedIn(false); showStatus('Erro ao verificar administrador: ' + result.error.message); return false; } if (!result.data) { setLoggedIn(false); showStatus('Usuário não autorizado na tabela admin_users.'); return false; } setLoggedIn(true); showStatus('Acesso administrativo confirmado.'); const categoriesLoaded = await loadCategories(); await Promise.all([ loadProducts(), loadFees() ]); if (!categoriesLoaded) { showStatus('Confira se existem categorias ativas cadastradas.'); } return true; } $('loginForm').addEventListener('submit', async event => { event.preventDefault(); showStatus('Verificando login...'); const { error } = await db.auth.signInWithPassword({ email: $('email').value.trim(), password: $('password').value }); if (error) { showStatus('Falha no login: ' + error.message); return; } await verifyAdmin(); }); // Upload de imagem para o Supabase Storage async function uploadImage(file) { if (!file) return currentImageUrl; if (!file.type.startsWith('image/')) { throw new Error('Selecione um arquivo de imagem válido.'); } if (file.size > MAX_IMAGE_SIZE) { throw new Error('A imagem deve ter no máximo 5 MB.'); } const allowedTypes = [ 'image/jpeg', 'image/png', 'image/webp', 'image/gif' ]; if (!allowedTypes.includes(file.type)) { throw new Error('Use uma imagem JPG, PNG, WEBP ou GIF.'); } const extensionMap = { 'image/jpeg': 'jpg', 'image/png': 'png', 'image/webp': 'webp', 'image/gif': 'gif' }; const extension = extensionMap[file.type]; const uniqueId = crypto.randomUUID ? crypto.randomUUID() : Date.now() + '-' + Math.random().toString(36).slice(2); const path = 'products/' + uniqueId + '.' + extension; const { error } = await db.storage .from(BUCKET) .upload(path, file, { cacheControl: '3600', upsert: false, contentType: file.type }); if (error) { throw new Error( 'Falha no upload: ' + error.message + '. Verifique as políticas de Storage do bucket produtos.' ); } const { data } = db.storage .from(BUCKET) .getPublicUrl(path); return data.publicUrl; } // Carregar e exibir produtos async function loadProducts() { const { data, error } = await db .from('products') .select('*') .order('id', { ascending: false }); if (error) { showStatus('Erro ao carregar produtos: ' + error.message); return; } productsCache = data || []; renderProducts(); } function renderProducts() { const root = $('products'); root.replaceChildren(); const selected = $('filterCategory').value; const list = productsCache.filter(product => !selected || String(product.category_id || '') === selected ); if (!list.length) { addText( root, 'p', productsCache.length ? 'Nenhum produto encontrado nesta categoria.' : 'Nenhum produto cadastrado ainda.' ); return; } list.forEach(product => { const item = document.createElement('div'); item.className = 'item'; if (product.image_url) { const image = document.createElement('img'); image.src = product.image_url; image.alt = product.name || 'Foto do produto'; image.loading = 'lazy'; image.style.cssText = 'width:110px;height:110px;object-fit:cover;border-radius:8px;margin-bottom:8px;'; item.appendChild(image); } addText( item, 'strong', product.name + ' — ' + money(product.price) ); addText( item, 'div', 'Categoria: ' + categoryName(product.category_id) ).className = 'tag'; addText( item, 'p', (product.available ? 'Disponível' : 'Indisponível') + (product.description ? ' · ' + product.description : '') ); const edit = addText(item, 'button', 'Editar'); edit.onclick = () => { $('productId').value = product.id; $('name').value = product.name || ''; $('description').value = product.description || ''; $('price').value = product.price; $('available').checked = !!product.available; $('category').value = product.category_id ? String(product.category_id) : ''; currentImageUrl = product.image_url || ''; $('imageFile').value = ''; // Mantém compatibilidade com o campo URL existente no HTML. if ($('image')) $('image').value = currentImageUrl; resetPreview(); $('productForm').scrollIntoView({ behavior: 'smooth' }); showStatus('Editando produto: ' + product.name); }; const toggle = addText( item, 'button', product.available ? 'Desativar' : 'Ativar' ); toggle.className = 'secondary'; toggle.onclick = () => saveProduct({ ...product, available: !product.available }); const remove = addText(item, 'button', 'Excluir'); remove.className = 'danger'; remove.onclick = () => deleteProduct(product); root.appendChild(item); }); } // Salvar produto: com foto nova ou mantendo a foto atual async function saveProduct(product) { try { const file = $('imageFile').files[0]; let imageUrl = product.image_url || currentImageUrl || ''; if (file) { showStatus('Enviando foto para o Supabase...'); imageUrl = await uploadImage(file); } const payload = { name: String(product.name || '').trim(), description: String(product.description || '').trim(), price: Number(product.price), image_url: imageUrl, available: !!product.available, category_id: product.category_id || null }; if (!payload.name) { throw new Error('Informe o nome do produto.'); } if (!Number.isFinite(payload.price) || payload.price < 0) { throw new Error('Informe um preço válido.'); } showStatus('Salvando produto...'); const result = product.id ? await db.from('products') .update(payload) .eq('id', product.id) : await db.from('products').insert(payload); if (result.error) throw result.error; clearProductForm(); await loadProducts(); showStatus('Produto salvo com sucesso.'); } catch (error) { showStatus('Erro ao salvar produto: ' + error.message); } } $('productForm').addEventListener('submit', async event => { event.preventDefault(); if (!$('category').value) { showStatus('Selecione uma categoria antes de salvar.'); $('category').focus(); return; } await saveProduct({ id: $('productId').value || null, name: $('name').value.trim(), description: $('description').value.trim(), price: $('price').value, image_url: currentImageUrl, available: $('available').checked, category_id: $('category').value }); }); // Prévia da foto selecionada $('imageFile').addEventListener('change', () => { const file = $('imageFile').files[0]; if (!file) { resetPreview(); return; } const allowedTypes = [ 'image/jpeg', 'image/png', 'image/webp', 'image/gif' ]; if (!allowedTypes.includes(file.type)) { $('imageFile').value = ''; showStatus('Use uma imagem JPG, PNG, WEBP ou GIF.'); return; } if (file.size > MAX_IMAGE_SIZE) { $('imageFile').value = ''; showStatus('A imagem deve ter no máximo 5 MB.'); return; } if (previewUrl) URL.revokeObjectURL(previewUrl); previewUrl = URL.createObjectURL(file); $('imagePreview').src = previewUrl; $('imagePreview').classList.remove('hidden'); showStatus('Foto selecionada. Clique em Salvar produto para enviar.'); }); $('clearProduct').onclick = clearProductForm; $('filterCategory').addEventListener('change', renderProducts); async function deleteProduct(product) { if (!confirm('Excluir o produto "' + product.name + '"?')) return; const { error } = await db .from('products') .delete() .eq('id', product.id); if (error) { showStatus('Erro ao excluir produto: ' + error.message); return; } showStatus('Produto excluído.'); await loadProducts(); } // Taxas de entrega async function loadFees() { const { data, error } = await db .from('delivery_fees') .select('*') .order('neighborhood'); if (error) { showStatus('Erro ao carregar taxas: ' + error.message); return; } feesCache = data || []; renderFees(); } function renderFees() { const root = $('fees'); root.replaceChildren(); if (!feesCache.length) { addText(root, 'p', 'Nenhuma taxa cadastrada ainda.'); return; } feesCache.forEach(fee => { const item = document.createElement('div'); item.className = 'item'; addText( item, 'strong', fee.neighborhood + ' — ' + money(fee.fee) ); addText( item, 'p', fee.active ? 'Região atendida' : 'Região desativada' ); const edit = addText(item, 'button', 'Editar'); edit.onclick = () => { $('feeId').value = fee.id; $('neighborhood').value = fee.neighborhood; $('fee').value = fee.fee; $('active').checked = !!fee.active; $('feeForm').scrollIntoView({ behavior: 'smooth' }); }; const toggle = addText( item, 'button', fee.active ? 'Desativar' : 'Ativar' ); toggle.className = 'secondary'; toggle.onclick = () => saveFee({ ...fee, active: !fee.active }); const remove = addText(item, 'button', 'Excluir'); remove.className = 'danger'; remove.onclick = () => deleteFee(fee); root.appendChild(item); }); } async function saveFee(fee) { const payload = { neighborhood: String(fee.neighborhood || '').trim(), fee: Number(fee.fee), active: !!fee.active }; if (!payload.neighborhood) { showStatus('Informe o bairro ou setor.'); return; } if (!Number.isFinite(payload.fee) || payload.fee < 0) { showStatus('Informe uma taxa válida.'); return; } showStatus('Salvando taxa...'); const result = fee.id ? await db.from('delivery_fees') .update(payload) .eq('id', fee.id) : await db.from('delivery_fees').insert(payload); if (result.error) { showStatus('Erro ao salvar taxa: ' + result.error.message); return; } clearFeeForm(); await loadFees(); showStatus('Taxa salva com sucesso.'); } $('feeForm').addEventListener('submit', async event => { event.preventDefault(); await saveFee({ id: $('feeId').value || null, neighborhood: $('neighborhood').value.trim(), fee: $('fee').value, active: $('active').checked }); }); async function deleteFee(fee) { if (!confirm('Excluir a taxa de "' + fee.neighborhood + '"?')) return; const { error } = await db .from('delivery_fees') .delete() .eq('id', fee.id); if (error) { showStatus('Erro ao excluir taxa: ' + error.message); return; } await loadFees(); showStatus('Taxa excluída.'); } $('clearFee').onclick = clearFeeForm; $('logout').onclick = async () => { const { error } = await db.auth.signOut(); if (error) { showStatus('Erro ao sair: ' + error.message); return; } setLoggedIn(false); clearProductForm(); clearFeeForm(); showStatus('Você saiu do painel.'); }; try { await verifyAdmin(); } catch (error) { showStatus('Erro inesperado: ' + error.message); } })();
